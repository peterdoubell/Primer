from pathlib import Path
import json,gzip,hashlib,numpy as np
D=Path(__file__).resolve().parent
p=json.loads((D/'reader-proof/browser-proof.json').read_bytes())
assert p['allPassed']
m=json.loads((D/'package/manifest.json').read_bytes())
def plane(v,axis,index):
 r=m['volumes'][v]['planes'][str(axis)][index]
 with (D/'package'/r['file']).open('rb')as f:f.seek(r['offset']);raw=gzip.decompress(f.read(r['gzipBytes']))
 return np.frombuffer(raw,np.uint8).reshape(r['shape'],order='F').T
count=0
for case in p['cases']:
 for row in case['planes']:
  a=row['axis'];l=plane('labels',a,case['ijk'][a]);hit=l==case['code']
  grey=plane('MRI',a,case['ijk'][a])if case['mode']=='mri'else np.where(l,100,20).astype(np.uint8)
  rgba=np.empty((*grey.shape,4),np.uint8);rgba[:,:,:3]=grey[:,:,None];rgba[:,:,3]=255;rgba[hit]=[20,180,175,255]
  if a==2:rgba=rgba[::-1]
  if a==0:rgba=rgba[:,::-1]
  assert hashlib.sha256(rgba.tobytes()).hexdigest()==row['sha'],(case['code'],case['mode'],a)
  count+=rgba.shape[0]*rgba.shape[1]
report={'allPassed':True,'cases':len(p['cases']),'planes':len(p['cases'])*3,'pixelsCompared':count,'browserProofSHA256':hashlib.sha256((D/'reader-proof/browser-proof.json').read_bytes()).hexdigest(),'clinicalApproval':False}
(D/'reader-proof/pixel-proof.json').write_text(json.dumps(report,indent=2)+'\n');print(report)
