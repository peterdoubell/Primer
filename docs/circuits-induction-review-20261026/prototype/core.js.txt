const topologies={series:{nodes:['A','B','C','D'],edges:[['A','B'],['B','C'],['C','D']]},parallel:{nodes:['A','D'],edges:[['A','D'],['A','D'],['A','D']]},mixed:{nodes:['A','B','D'],edges:[['A','B'],['B','D'],['B','D']]},bridge:{nodes:['A','B','C','D'],edges:[['A','B'],['B','D'],['A','C'],['C','D'],['B','C']]},single:{nodes:['A','D'],edges:[['A','D']]}};
function number(v,name,lo,hi){if(typeof v!=='number'||!Number.isFinite(v)||v<lo||v>hi)throw new RangeError(name+' must be finite in ['+lo+', '+hi+'].');return v;}
function decimalSum(values){
 const parts=values.map(value=>{const [mantissa,exponent='0']=value.toString().split('e'),negative=mantissa.startsWith('-'),digits=mantissa.replace('-','').split('.');return {coefficient:BigInt(digits.join(''))*(negative?-1n:1n),power:Number(exponent)-(digits[1]||'').length};});

}
function linear(A,b){const n=b.length,M=A.map((r,i)=>[...r,b[i]]);for(let k=0;k<n;k++){let pivot=k;for(let i=k+1;i<n;i++)if(Math.abs(M[i][k])>Math.abs(M[pivot][k]))pivot=i;if(Math.abs(M[pivot][k])<1e-14)throw Error('Singular reduced circuit matrix.');[M[k],M[pivot]]=[M[pivot],M[k]];const scale=M[k][k];for(let j=k;j<=n;j++)M[k][j]/=scale;for(let i=0;i<n;i++)if(i!==k){const ratio=M[i][k];for(let j=k;j<=n;j++)M[i][j]-=ratio*M[k][j];}}return M.map(r=>r[n]);}

 const unknown=groups.filter(g=>reached.has(g)&&!Object.hasOwn(fixed,g)),A=unknown.map(()=>unknown.map(()=>0)),b=unknown.map(()=>0);

 // A short-edge bridge has a unique cut current. A short-edge cycle has a
 // free circulation and must not receive an invented equal split.

  if(!side.has(edge.to)&&!(freeSource&&side.has('A')!==side.has('D')))edge.current=-[...side].reduce((sum,n)=>sum+external[n],0);
 }

 const sourcePower=voltage*sourceCurrent,loadPower=result.reduce((s,e)=>s+e.power,0);
 return {status:freeSource?'underdetermined-source-loop':result.some(e=>e.current===null)?'underdetermined-wire-circulation':reached.size<groups.length?'solved-with-floating-nodes':'solved',voltage,edges:result,nodes:Object.fromEntries(nodes.map(n=>[n,V(n)])),sourceCurrent:freeSource?null:sourceCurrent,sourcePower,loadPower,powerResidual:sourcePower-loadPower,kcl,explanation:(freeSource?'An ideal zero-resistance path across the zero-volt source leaves its circulating current undetermined. }

function induction(raw={}){
